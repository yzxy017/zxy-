<script>
/* ===== 状态与渲染 ===== */
const state = { i: 0, answers: [] };
const $ = (s, r) => (r || document).querySelector(s);
const CUR = { id: 0 };

function showView(id) {
  document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
  $('#' + id).classList.add('active');
  window.scrollTo({ top: 0, behavior: 'auto' });
}

/* 进度刻度 */
function buildTicks() {
  const box = $('#ticks');
  box.innerHTML = QS.map(() => '<i></i>').join('');
}
function paintTicks() {
  const box = $('#ticks');
  [...box.children].forEach((el, k) => {
    el.className = k < state.i ? 'done' : (k === state.i ? 'cur' : '');
  });
  $('#cur-num').textContent = String(state.i + 1).padStart(2, '0');
}

/* 单题渲染 */
const LETTERS = ['A', 'B', 'C', 'D'];
function renderQ() {
  const q = QS[state.i];
  const wrap = $('#qwrap');
  wrap.innerHTML =
    '<div class="qno">第 ' + String(state.i + 1).padStart(2, '0') + ' 题</div>' +
    '<h2 class="qtitle">' + q.q + '</h2>' +
    '<div class="opts">' + q.o.map((o, k) =>
      '<button class="opt" data-k="' + k + '">' +
        '<span class="mark">' + LETTERS[k] +
          '<svg viewBox="0 0 12 12" aria-hidden="true"><path d="M1.4 6.2l2.7 2.8 6.4-7" stroke="#F4F2E2" stroke-width="1.4" fill="none" stroke-linecap="round"/></svg>' +
        '</span>' +
        '<span>' + o.x + '</span>' +
      '</button>'
    ).join('') + '</div>';
  wrap.querySelectorAll('.opt').forEach(b => {
    b.addEventListener('click', () => choose(+b.dataset.k));
  });
  if (state.answers[state.i] !== undefined) {
    const c = wrap.querySelector('.opt[data-k="' + state.answers[state.i] + '"]');
    if (c) c.classList.add('chosen');
  }
  paintTicks();
}

/* 选择 */
let lock = false;
function choose(k) {
  if (lock) return;
  lock = true;
  const wrap = $('#qwrap');
  state.answers[state.i] = k;
  wrap.querySelectorAll('.opt').forEach(b => b.classList.remove('chosen'));
  wrap.querySelector('.opt[data-k="' + k + '"]').classList.add('chosen');
  paintTicks();
  setTimeout(() => {
    lock = false;
    if (state.i < QS.length - 1) {
      state.i++;
      const w = $('#qwrap');
      w.style.opacity = '0';
      w.style.transform = 'translateY(8px)';
      w.style.transition = 'opacity .16s ease,transform .16s ease';
      setTimeout(() => {
        renderQ();
        w.style.transition = 'none';
        w.style.opacity = '1';
        w.style.transform = 'none';
      }, 170);
    } else {
      finish();
    }
  }, 230);
}

/* 计分：主分 + 核心题加权（同分时用核心题命中数排序） */
const CORE = [0, 4, 5, 10];
function tally() {
  const s = new Array(RES.length).fill(0);
  const core = new Array(RES.length).fill(0);
  state.answers.forEach((k, qi) => {
    const t = QS[qi].o[k].t;
    s[t] += 1;
    if (CORE.includes(qi)) core[t] += 1;
  });
  return s.map((v, idx) => ({ idx, v, core: core[idx] }))
          .sort((a, b) => b.v - a.v || b.core - a.core || a.idx - b.idx);
}

/* 结果页 */
function finish() {
  const rows = tally();
  const top = rows[0];
  const r = RES[top.idx];
  CUR.id = top.idx;
  const others = rows.slice(1, 3);
  const maxV = rows[0].v || 1;
  $('#view-result').style.setProperty('--c', r.c);

  const all = RES.map((x, k) =>
    '<button class="mini ' + (k === top.idx ? 'sel' : '') + '" data-k="' + k + '" style="--mc:' + x.c + '">' +
      '<strong>' + x.name + '</strong>' +
      '<small>' + x.tag + '</small>' +
      (k === top.idx ? '<span class="you">你的结果</span>' : '') +
    '</button>'
  ).join('');

  $('#view-result').innerHTML =
  '<div class="res-head" style="--c:' + r.c + '">' +
    '<p class="persona">你的定居人格 · ' + r.tag + '</p>' +
    '<h2 class="rname">' + r.name + '</h2>' +
    '<div class="cities">' + r.cities.map(c => '<span class="city">' + c + '</span>').join('') + '</div>' +
  '</div>' +
  '<p class="match-note">32 道题里，有 <b>' + top.v + '</b> 道指向这类生活。你的第二、第三顺位分别是「' +
    RES[others[0].idx].name.replace('型','') + '」和「' + RES[others[1].idx].name.replace('型','') + '」。</p>' +

  '<div class="blk"><h3><i></i>生活适配解读</h3>' + r.desc.map(t => '<p>' + t + '</p>').join('') + '</div>' +

  '<div class="blk"><h3><i></i>定居优势</h3><ul class="perks">' +
    r.perks.map(t => '<li><svg viewBox="0 0 24 24" aria-hidden="true"><path d="M4 13l5 5 11-12" stroke-width="1.8" stroke-linecap="round" fill="none"/></svg><span>' + t + '</span></li>').join('') +
  '</ul></div>' +

  '<div class="blk"><h3><i></i>适配人群</h3><p>' + r.who + '</p></div>' +

  '<div class="vibe"><span class="lab">生活氛围感</span><p>' + r.vibe + '</p></div>' +

  '<div class="blk"><h3><i></i>与你的契合顺位</h3><div class="rank">' +
    [top].concat(others).map((o, n) =>
      '<div class="rank-row"><div class="rank-top"><span>' + (n === 0 ? '第一顺位' : '第' + ['一','二','三','四'][n] + '顺位') + ' · ' + RES[o.idx].name + '</span><span>' + o.v + ' / 32</span></div>' +
      '<div class="track"><b style="--w:' + Math.max(12, Math.round(o.v / maxV * 100)) + '%;--rc:' + RES[o.idx].c + '"></b></div></div>'
    ).join('') +
  '</div></div>' +

  '<div class="blk"><h3><i></i>10 种定居人格全览</h3><div class="all-grid">' + all + '</div></div>' +

  '<div class="actions">' +
    '<button class="btn btn-main" id="btn-copy">复制我的结果文案</button>' +
    '<button class="btn btn-line" id="btn-again">重新测一次</button>' +
    '<p class="share-tip">截图这一页发小红书，比发链接更容易被点开</p>' +
  '</div>' +

  '<div class="foot"><b>趣味测试仅供娱乐，愿你奔赴属于自己的山海与温柔</b>本测试不含任何地域评价与人身指向，仅以生活方式偏好为线索；十种类型没有高下之分，所有城市都值得被喜欢。</div>';

  $('#btn-again').addEventListener('click', reset);
  $('#btn-copy').addEventListener('click', copyResult);
  $('#view-result').querySelectorAll('.mini').forEach(b => {
    b.addEventListener('click', () => preview(+b.dataset.k));
  });
  showView('view-result');
}

/* 全览里预览其它人格 */
function preview(k) {
  const r = RES[k];
  CUR.id = k;
  const view = $('#view-result');
  view.querySelectorAll('.mini').forEach(b => b.classList.toggle('sel', +b.dataset.k === k));
  const head = view.querySelector('.res-head');
  view.style.setProperty('--c', r.c);
  head.style.setProperty('--c', r.c);
  head.querySelector('.persona').textContent = '你的定居人格 · ' + r.tag;
  head.querySelector('.rname').textContent = r.name;
  head.querySelector('.cities').innerHTML = r.cities.map(c => '<span class="city">' + c + '</span>').join('');
  const blks = view.querySelectorAll('.blk');
  blks[0].innerHTML = '<h3><i></i>生活适配解读</h3>' + r.desc.map(t => '<p>' + t + '</p>').join('');
  blks[1].innerHTML = '<h3><i></i>定居优势</h3><ul class="perks">' +
    r.perks.map(t => '<li><svg viewBox="0 0 24 24" aria-hidden="true"><path d="M4 13l5 5 11-12" stroke-width="1.8" stroke-linecap="round" fill="none" stroke="' + r.c + '"/></svg><span>' + t + '</span></li>').join('') + '</ul>';
  blks[2].innerHTML = '<h3><i></i>适配人群</h3><p>' + r.who + '</p>';
  view.querySelector('.vibe p').textContent = r.vibe;
  view.scrollIntoView({ block: 'start', behavior: 'smooth' });
}

/* 复制文案 */
function copyResult() {
  const r = RES[CUR.id];
  const rows = tally();
  const mine = rows.find(x => x.idx === CUR.id);
  const text = '我在「测一测你适合在哪个城市定居」里测出来是「' + r.name + '」——' + r.tag + '。\n' +
    '代表城市：' + r.cities.join('、') + '\n' +
    r.vibe + '\n' +
    '32 道题里，有 ' + (mine ? mine.v : 0) + ' 道指向这种生活。\n' +
    '#城市定居测试 #治愈系测试 #宜居城市 #生活方式';
  const done = () => toast('已复制，去发小红书吧');
  if (navigator.clipboard && window.isSecureContext) {
    navigator.clipboard.writeText(text).then(done, fallback);
  } else { fallback(); }
  function fallback() {
    const ta = document.createElement('textarea');
    ta.value = text; ta.style.position = 'fixed'; ta.style.opacity = '0';
    document.body.appendChild(ta); ta.select();
    try { document.execCommand('copy'); done(); } catch (e) { toast('复制失败，长按屏幕手动保存'); }
    document.body.removeChild(ta);
  }
}
let toastTimer;
function toast(msg) {
  const t = $('#toast');
  t.textContent = msg; t.classList.add('on');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => t.classList.remove('on'), 1900);
}

/* 重置 */
function reset() {
  state.i = 0; state.answers = [];
  buildTicks(); renderQ(); showView('view-quiz');
}

/* 启动 */
$('#btn-start').addEventListener('click', reset);
$('#btn-back').addEventListener('click', () => {
  if (state.i === 0) { showView('view-home'); return; }
  state.i--; renderQ();
});
buildTicks();
paintTicks();
</script>
